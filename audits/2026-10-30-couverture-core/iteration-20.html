Progrès et limites

Moins incertain : Les traces, leur qualité et les indicateurs ont des contrats séparés. Les inconnus, dossiers ouverts, corrections et branches parallèles sont pris en compte. Le contrôle final donne un propriétaire de financement à chaque fonction native et aux raccords transverses restant à qualifier.

Reste incertain : Process Mining ne dispose pas ici d'une référence produit complète équivalente dans les trois ERP ; la réalisation Java est estimée par décomposition. Identité/corrélation des faits, choix des assiettes et performance multi-domaines restent sensibles. Toutes les charges demeurent non calibrées.

    def _select_dict(self, table: TableSQL):

        return from_item.select(*cls._columns(tables, withs),

    @classmethod
    def _sale_line(cls, length, index, company_id=None):
        pool = Pool()
        Line = pool.get('sale.line')
        Sale = pool.get('sale.sale')

        line = Line.__table__()
        sale = Sale.__table__()

        return (line
            .join(sale, condition=line.sale == sale.id)
            .select(
                (line.id * length + index).as_('id'),
                *cls._sale_line_columns(line, sale),
                where=sale.state.in_(cls._sale_states())
                & (sale.company == company_id),
                ))

    def history_revisions(cls, ids):
        pool = Pool()
        ModelAccess = pool.get('ir.model.access')
        User = pool.get('res.user')
        cursor = Transaction().connection.cursor()

        ModelAccess.check(cls.__name__, 'read')

        table = cls.__table_history__()
        user = User.__table__()
        columns = [
            Coalesce(table.write_date, table.create_date), table.id, user.name]
        where = fields.SQL_OPERATORS['in'](table.id, ids)
        cursor.execute(*table.join(user, 'LEFT',
                Coalesce(table.write_uid, table.create_uid) == user.id)
            .select(*columns, where=where, group_by=columns))
        revisions = list(cursor)
        revisions.sort(reverse=True)
        # SQLite uses char for COALESCE
        if revisions and isinstance(revisions[0][0], str):
            strptime = datetime.datetime.strptime
            format_ = '%Y-%m-%d %H:%M:%S.%f'
            revisions = [(strptime(timestamp, format_), id_, name)
                for timestamp, id_, name in revisions]
        return revisions

  protected String prepareQuery(CrmReporting crmReporting, boolean isPartner, String model) {
    StringBuilder query = new StringBuilder();
    model = Strings.isNullOrEmpty(model) ? "" : model + ".";

    if (isPartner) {
      partnerQuery(query, crmReporting, model);
    } else {
      leadQuery(query, crmReporting, model);
    }

    if (!crmReporting.getAgencySet().isEmpty()
        && ((AppCrm) appBaseService.getApp("crm")).getAgenciesManagement())
      this.addParams(
          query,
          "self."
              + model
              + "agency.id IN ("
              + StringHelper.getIdListString(crmReporting.getAgencySet())
              + ")");

    if (!crmReporting.getIndustrySectorSet().isEmpty())
      this.addParams(
          query,
          "self."
              + model
              + "industrySector.id IN ("
              + StringHelper.getIdListString(crmReporting.getIndustrySectorSet())
              + ")");

    if (appBaseService.getAppBase().getTeamManagement() && !crmReporting.getTeamSet().isEmpty())
      this.addParams(
          query,
          "self.team.id IN (" + StringHelper.getIdListString(crmReporting.getTeamSet()) + ")");

    if (crmReporting.getFromDate() != null)
      this.addParams(query, "date(self.createdOn) >= :fromDate");

    if (crmReporting.getToDate() != null) this.addParams(query, "date(self.createdOn) <= :toDate");

    return query.toString();
  }
